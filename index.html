Как подключить

Шаги настройки

  1. Создайте проект Supabase и выполните SQL-схему из первого шага.
  2. Откройте Project Settings → API и скопируйте Project URL и anon public key.
  3. Вставьте URL и anon key в форму выше или в блок window.SYSTEMHUB_SUPABASE_*.
  4. Для быстрого теста оставьте временные RLS-политики public read/write.
  5. Позже замените public-политики на Supabase Auth и роли.
  6. Откройте этот HTML-файл или разверните его на статическом хостинге.
create table if not exists public.app_users (
  id text primary key,
  login text not null unique,
  "displayName" text,
  role text not null default 'user',
  "passwordHash" text not null,
  "passwordSalt" text not null,
  "isActive" boolean not null default true,
  "createdAt" timestamptz not null default now(),
  "updatedAt" timestamptz not null default now()
);

alter table public.app_users enable row level security;

create policy "systemhub app users read"
on public.app_users for select
using (true);

create policy "systemhub app users write"
on public.app_users for all
using (true)
with check (true);

create table if not exists public.app_settings (
  key text primary key,
  value jsonb not null default '{}'::jsonb,
  "updatedAt" timestamptz not null default now(),
  "updatedBy" text
);

alter table public.app_settings enable row level security;

create policy "systemhub app settings read"
on public.app_settings for select
using (true);

create policy "systemhub app settings write"
on public.app_settings for all
using (true)
with check (true);

grant usage on schema public to anon, authenticated;
grant select, insert, update, delete on public.app_settings to anon, authenticated;
notify pgrst, 'reload schema';

Supabase теперь используется как основной источник данных. Изменения сохраняются по разделам и автоматически подтягиваются на других устройствах. Большие загруженные отчеты остаются файловыми, чтобы не перегружать браузер.
